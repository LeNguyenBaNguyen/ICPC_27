def addini(
        self,
        name: str,
        help: str,
        type: Literal["string", "paths", "pathlist", "args", "linelist", "bool"]
        | None = None,
        default: Any = NOT_SET,
    ) -> None:































        assert type in (None, "string", "paths", "pathlist", "args", "linelist", "bool")

        self._inidict[name] = (help, type, default)
        self._ininames.append(name)
